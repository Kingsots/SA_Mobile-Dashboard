# Non-destructive: does not modify existing generate_signals() method

from typing import Optional, Dict
from signals.event_filter import MarketEvent
import pandas as pd
import numpy as np
import logging

logger = logging.getLogger(__name__)


def generate_signal_on_event(self, ticker: str, interval: str, 
                             event: MarketEvent) -> Optional[Dict]:
    """
    Generate signal triggered by market event (event-driven mode)
    
    Args:
        ticker: Symbol ticker
        interval: Timeframe
        event: MarketEvent that triggered this signal generation
        
    Returns:
        Signal dict or None if no signal
    """
    try:
        logger.info(
            f"Generating event-driven signal: {ticker} {interval} "
            f"triggered by {event.type} (confidence: {event.confidence:.2f})"
        )
        
        # 1. Get latest features (reuse existing method)
        features = self.get_latest_features(ticker, interval)
        
        if features is None or features.empty:
            logger.warning(f"No features available for {ticker} {interval}")
            return None
        
        # 2. Coerce numeric features (handle any NaN or non-numeric values)
        for col in features.columns:
            if col not in ['ticker', 'interval', 'timestamp']:
                features[col] = pd.to_numeric(features[col], errors='coerce').fillna(0)
        
        # 3. Get ML prediction (reuse existing predict_signal method)
        signal_result = self.predict_signal(ticker, interval, features)
        
        if signal_result is None:
            logger.warning(f"No signal generated for {ticker} {interval}")
            return None
        
        # 4. Extract signal and confidence
        predicted_signal = signal_result.get('signal')
        ml_confidence = signal_result.get('confidence', 0.0)
        
        # 5. Check event-signal alignment
        event_direction = self._infer_event_direction(event)
        signal_matches_event = self._check_signal_event_alignment(
            predicted_signal, event_direction
        )
        
        # 6. Adjust confidence based on alignment
        if not signal_matches_event:
            logger.warning(
                f"Event-signal mismatch: event={event_direction}, "
                f"signal={predicted_signal}"
            )
            # Lower confidence if they contradict
            ml_confidence = ml_confidence * 0.7
            contradict_event = True
        else:
            # Boost confidence if they align
            contradict_event = False
            # Combine event confidence with ML confidence
            combined_confidence = (ml_confidence + event.confidence) / 2
            ml_confidence = min(combined_confidence, 0.95)
        
        # 7. Build enhanced signal dict
        signal_dict = {
            'ticker': ticker,
            'interval': interval,
            'signal': predicted_signal,
            'confidence': ml_confidence,
            'timestamp': pd.Timestamp.now(),
            
            # Event context
            'triggered_by': 'EVENT',
            'event_type': event.type,
            'event_confidence': event.confidence,
            'event_details': event.details,
            'event_ts': event.ts,
            'contradict_event': contradict_event,
            
            # Original ML features/predictions
            **signal_result
        }
        
        # 8. Calculate trade levels (entry, SL, TP)
        trade_levels = self._calculate_trade_levels(
            ticker, interval, predicted_signal, features
        )
        signal_dict.update(trade_levels)
        
        logger.info(
            f"Event-driven signal generated: {ticker} {predicted_signal} "
            f"confidence={ml_confidence:.2f} event={event.type}"
        )
        
        return signal_dict
        
    except Exception as e:
        logger.error(
            f"Error generating event-driven signal for {ticker}: {e}",
            exc_info=True
        )
        return None


def _infer_event_direction(self, event: MarketEvent) -> str:
    """
    Infer bullish/bearish direction from event type
    
    Returns:
        'bullish', 'bearish', or 'neutral'
    """
    event_type = event.type.lower()
    
    # Bullish events
    if any(x in event_type for x in [
        'bullish', 'cross_bullish', 'break_support'
    ]):
        return 'bullish'
    
    # Bearish events
    if any(x in event_type for x in [
        'bearish', 'cross_bearish', 'break_resistance'
    ]):
        return 'bearish'
    
    # Neutral/ambiguous
    return 'neutral'


def _check_signal_event_alignment(self, signal: str, event_direction: str) -> bool:
    """
    Check if signal aligns with event direction
    
    Returns:
        True if aligned or neutral
    """
    if event_direction == 'neutral':
        return True
    
    signal_lower = str(signal).lower()
    
    if event_direction == 'bullish':
        return 'buy' in signal_lower or 'long' in signal_lower
    
    if event_direction == 'bearish':
        return 'sell' in signal_lower or 'short' in signal_lower
    
    return True


def _calculate_trade_levels(self, ticker: str, interval: str, 
                            signal: str, features: pd.DataFrame) -> Dict:
    """
    Calculate entry, stop loss, and take profit levels
    
    Returns:
        Dict with 'entry_price', 'stop_loss', 'take_profit', 'risk_reward'
    """
    try:
        # Get current price from features
        if 'close' in features.columns:
            current_price = float(features['close'].iloc[-1])
        else:
            logger.warning("No 'close' price in features")
            return {}
        
        # Get ATR for risk calculation
        atr = float(features.get('atr', 0)) if 'atr' in features.columns else 0
        
        # Fallback: use 2% of price as ATR if not available
        if atr == 0 or pd.isna(atr):
            atr = current_price * 0.02
        
        # Calculate levels based on signal direction
        signal_lower = str(signal).lower()
        
        if 'buy' in signal_lower or 'long' in signal_lower:
            # Long position
            entry_price = current_price
            stop_loss = current_price - (1.5 * atr)  # 1.5 ATR below entry
            take_profit = current_price + (3 * atr)   # 3 ATR above entry (2:1 RR)
            risk_reward = 2.0
            
        elif 'sell' in signal_lower or 'short' in signal_lower:
            # Short position
            entry_price = current_price
            stop_loss = current_price + (1.5 * atr)   # 1.5 ATR above entry
            take_profit = current_price - (3 * atr)    # 3 ATR below entry (2:1 RR)
            risk_reward = 2.0
            
        else:
            # Neutral or unknown
            return {
                'entry_price': current_price,
                'stop_loss': None,
                'take_profit': None,
                'risk_reward': None
            }
        
        return {
            'entry_price': round(entry_price, 2),
            'stop_loss': round(stop_loss, 2),
            'take_profit': round(take_profit, 2),
            'risk_reward': risk_reward,
            'atr_used': round(atr, 2)
        }
        
    except Exception as e:
        logger.error(f"Error calculating trade levels: {e}")
        return {}

"""

Run manually: python scripts/migrate_add_triggered_by.py

import asyncpg
import asyncio
import logging
from datetime import datetime

logging.basicConfig(level=logging.INFO)
logger = logging.getLogger(__name__)


async def run_migration():
    """Add triggered_by column to signals table"""
    
    # Load database config
    from config import DB_HOST, DB_PORT, DB_NAME, DB_USER, DB_PASSWORD
    
    conn = None
    try:
        # Connect to database
        conn = await asyncpg.connect(
            host=DB_HOST,
            port=DB_PORT,
            database=DB_NAME,
            user=DB_USER,
            password=DB_PASSWORD
        )
        
        logger.info("Connected to database")
        
        # Check if column already exists
        check_query = """
            SELECT column_name 
            FROM information_schema.columns 
            WHERE table_name = 'signals' 
            AND column_name = 'triggered_by'
        """
        
        existing = await conn.fetchval(check_query)
        
        if existing:
            logger.info("Column 'triggered_by' already exists. Skipping migration.")
            return
        
        # Add column with default value 'TIME'
        alter_query = """
            ALTER TABLE signals 
            ADD COLUMN triggered_by TEXT DEFAULT 'TIME'
        """
        
        logger.info("Adding 'triggered_by' column to signals table...")
        await conn.execute(alter_query)
        
        # Create index for performance
        index_query = """
            CREATE INDEX IF NOT EXISTS idx_signals_triggered_by 
            ON signals(triggered_by)
        """
        
        logger.info("Creating index on 'triggered_by' column...")
        await conn.execute(index_query)
        
        # Verify migration
        verify_query = """
            SELECT column_name, data_type, column_default
            FROM information_schema.columns 
            WHERE table_name = 'signals' 
            AND column_name = 'triggered_by'
        """
        
        result = await conn.fetchrow(verify_query)
        
        if result:
            logger.info(f"Migration successful: {dict(result)}")
        else:
            logger.error("Migration verification failed")
        
        # Add comment to column
        comment_query = """
            COMMENT ON COLUMN signals.triggered_by IS 
            'Signal trigger type: TIME (scheduled) or EVENT (event-driven)'
        """
        await conn.execute(comment_query)
        
        logger.info("Migration completed successfully")
        
    except Exception as e:
        logger.error(f"Migration failed: {e}", exc_info=True)
        raise
    
    finally:
        if conn:
            await conn.close()
            logger.info("Database connection closed")


async def rollback_migration():
    """Rollback migration (remove triggered_by column)"""
    
    from config import DB_HOST, DB_PORT, DB_NAME, DB_USER, DB_PASSWORD
    
    conn = None
    try:
        conn = await asyncpg.connect(
            host=DB_HOST,
            port=DB_PORT,
            database=DB_NAME,
            user=DB_USER,
            password=DB_PASSWORD
        )
        
        logger.info("Connected to database for rollback")
        
        # Drop index
        await conn.execute("DROP INDEX IF EXISTS idx_signals_triggered_by")
        
        # Drop column
        await conn.execute("ALTER TABLE signals DROP COLUMN IF EXISTS triggered_by")
        
        logger.info("Rollback completed successfully")
        
    except Exception as e:
        logger.error(f"Rollback failed: {e}", exc_info=True)
        raise
    
    finally:
        if conn:
            await conn.close()


if __name__ == "__main__":
    import sys
    
    if len(sys.argv) > 1 and sys.argv[1] == '--rollback':
        print("Running rollback...")
        asyncio.run(rollback_migration())
    else:
        print("Running migration...")
        print("Use --rollback flag to undo this migration")
        asyncio.run(run_migration())

# Event-Driven Signal Generation Implementation

## Overview

This implementation adds event-driven signal generation to your trading system while preserving the original time-based approach. The system now detects market events (trendline breaks, volume spikes, EMA crossovers, etc.) and generates signals only when significant market conditions occur.

## 🎯 Key Features

- **Event Detection**: 6 event types monitored (trendline breaks, volume spikes, EMA crossovers, volatility expansion, VWAP crosses, engulfing patterns)
- **Feature Flagged**: Enable/disable via `USE_EVENT_MONITOR` environment variable
- **Non-Destructive**: Original time-based job remains intact and functional
- **Cooldown Management**: Prevents duplicate signals for same event within configurable period
- **Portfolio Limits**: Configurable max active signals to manage risk
- **Signal Quality Tiers**: STRONG/MODERATE/WEAK based on confidence and rule matches
- **Multi-Timeframe Support**: Ready for anchor/execution timeframe confirmation

class Colors:
    """Terminal colors for output"""
    HEADER = '\033[95m'
    OKBLUE = '\033[94m'
    OKCYAN = '\033[96m'
    OKGREEN = '\033[92m'
    WARNING = '\033[93m'
    FAIL = '\033[91m'
    ENDC = '\033[0m'
    BOLD = '\033[1m'


def print_header(text):
    """Print colored header"""
    print(f"\n{Colors.HEADER}{Colors.BOLD}{'='*60}{Colors.ENDC}")
    print(f"{Colors.HEADER}{Colors.BOLD}{text.center(60)}{Colors.ENDC}")
    print(f"{Colors.HEADER}{Colors.BOLD}{'='*60}{Colors.ENDC}\n")


def print_success(text):
    """Print success message"""
    print(f"{Colors.OKGREEN}✓ {text}{Colors.ENDC}")


def print_warning(text):
    """Print warning message"""
    print(f"{Colors.WARNING}⚠ {text}{Colors.ENDC}")


def print_error(text):
    """Print error message"""
    print(f"{Colors.FAIL}✗ {text}{Colors.ENDC}")


def print_info(text):
    """Print info message"""
    print(f"{Colors.OKCYAN}ℹ {text}{Colors.ENDC}")


def generate_test_data(length=200):
    """Generate synthetic OHLCV data for testing"""
    dates = pd.date_range(start='2024-01-01', periods=length, freq='1H')
    
    # Create price trend with noise
    base_price = 100
    trend = np.linspace(0, 15, length)
    noise = np.random.randn(length) * 2
    close_prices = base_price + trend + noise
    
    high_prices = close_prices + np.random.rand(length) * 2
    low_prices = close_prices - np.random.rand(length) * 2
    open_prices = close_prices + np.random.randn(length) * 1
    
    # Volume with spikes
    volume = np.random.randint(1000000, 5000000, length)
    volume[int(length * 0.7)] = 10000000  # Add spike
    volume[int(length * 0.9)] = 15000000
    
    df = pd.DataFrame({
        'open': open_prices,
        'high': high_prices,
        'low': low_prices,
        'close': close_prices,
        'volume': volume
    }, index=dates)
    
    return df

def test_event_detection():
    """Test event detection functionality"""
    print_header("Event Detection Tests")
    
    # Generate test data
    df = generate_test_data(200)
    print_info(f"Generated {len(df)} bars of test data")
    
    # Initialize monitor
    monitor = EventMonitor()
    print_success("EventMonitor initialized")
    
    # Test event detection
    try:
        events = monitor.check_for_events('TEST', '1h', df)
        print_success(f"Detected {len(events)} events")
        
        # Display event types
        if events:
            event_types = {}
            for event in events:
                event_types[event.type] = event_types.get(event.type, 0) + 1
            
            print_info("Event breakdown:")
            for event_type, count in event_types.items():
                print(f"  - {event_type}: {count}")
        else:
            print_warning("No events detected (this may be normal for random data)")
        
        # Validate event structure
        for event in events:
            assert hasattr(event, 'type')
            assert hasattr(event, 'confidence')
            assert hasattr(event, 'details')
            assert hasattr(event, 'ts')
            assert 0 <= event.confidence <= 1.0
        
        print_success("All events have valid structure")
        return True
        
    except Exception as e:
        print_error(f"Event detection failed: {e}")
        return False


def test_cooldown_logic():
    """Test event cooldown filtering"""
    print_header("Cooldown Logic Tests")
    
    filter = EventFilter(cooldown_seconds=3600)
    print_success("EventFilter initialized")
    
    # Test first event
    event1 = MarketEvent(
        type='volume_spike',
        confidence=0.8,
        details={},
        ts=datetime.now(),
        ticker='TEST',
        interval='1h'
    )
    
    is_valid = filter.is_valid(event1, 'TEST')
    if is_valid:
        print_success("First event passed (as expected)")
    else:
        print_error("First event should be valid")
        return False
    
    # Test duplicate
    event2 = MarketEvent(
        type='volume_spike',
        confidence=0.8,
        details={},
        ts=datetime.now(),
        ticker='TEST',
        interval='1h'
    )
    
    is_valid = filter.is_valid(event2, 'TEST')
    if not is_valid:
        print_success("Duplicate event filtered (as expected)")
    else:
        print_error("Duplicate event should be filtered")
        return False
    
    # Test different ticker
    event3 = MarketEvent(
        type='volume_spike',
        confidence=0.8,
        details={},
        ts=datetime.now(),
        ticker='OTHER',
        interval='1h'
    )
    
    is_valid = filter.is_valid(event3, 'OTHER')
    if is_valid:
        print_success("Different ticker event passed (as expected)")
    else:
        print_error("Different ticker should be valid")
        return False
    
    return True

def generate_report(results):
    """Generate final validation report"""
    print_header("Validation Report")
    
    total_tests = len(results)
    passed_tests = sum(1 for r in results.values() if r)
    failed_tests = total_tests - passed_tests
    
    print(f"Total Tests: {total_tests}")
    print(f"{Colors.OKGREEN}Passed: {passed_tests}{Colors.ENDC}")
    
    if failed_tests > 0:
        print(f"{Colors.FAIL}Failed: {failed_tests}{Colors.ENDC}")
    else:
        print(f"{Colors.OKGREEN}Failed: {failed_tests}{Colors.ENDC}")
    
    print("\nDetailed Results:")
    for test_name, result in results.items():
        status = f"{Colors.OKGREEN}PASS{Colors.ENDC}" if result else f"{Colors.FAIL}FAIL{Colors.ENDC}"
        print(f"  {test_name}: {status}")
    
    print("\n" + "="*60)
    
    if failed_tests == 0:
        print(f"\n{Colors.OKGREEN}{Colors.BOLD}✓ ALL TESTS PASSED - SYSTEM READY{Colors.ENDC}")
        print(f"\n{Colors.OKCYAN}Next steps:{Colors.ENDC}")
        print(f"  1. Set USE_EVENT_MONITOR=true in your environment")
        print(f"  2. Run: python scripts/migrate_add_triggered_by.py")
        print(f"  3. Restart scheduler: python async_scheduler.py")
        print(f"  4. Monitor logs: tail -f logs/signal_debug.log")
        return True
    else:
        print(f"\n{Colors.FAIL}{Colors.BOLD}✗ SOME TESTS FAILED{Colors.ENDC}")
        print(f"\n{Colors.WARNING}Please fix issues before deploying{Colors.ENDC}")
        return False
