import re
path = "signals/xgb_signal_engine.py"
txt = open(path).read()

pattern = "features_snapshot = df_features.iloc"
if pattern not in txt:
    print("❌ Pattern not found — STOP. Share file.")
else:
    patch = r"""
        # --- LEAKAGE PREVENTION PATCH ---
        # Drop suspicious/leaking columns before feature snapshot
        leak_cols = [
            'timestamp','created_at','date',
            'next_close','next_open','next_return','future_return',
        ]
        # Drop any column that starts with lead_ or target_next_
        leak_cols += [c for c in df_features.columns if c.startswith('lead_') or c.startswith('target_next_')]
        df_features = df_features.drop(columns=leak_cols, errors='ignore')

        # Keep ONLY numeric features (defensive)
        df_features = df_features.select_dtypes(include=['number'])
        # ---------------------------------
    """

    new = re.sub(r"(features_snapshot\s*=\s*df_features\.iloc\[-1\]\.to_dict\(\))",
                 patch + r"\n        \1", txt)

    open(path, "w").write(new)
    print("✅ Leakage Patch Applied: drop suspicious columns")
